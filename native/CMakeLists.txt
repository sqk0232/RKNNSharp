cmake_minimum_required(VERSION 3.16)
project(rknnsharp_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RKNN_INCLUDE_DIR "" CACHE PATH "Directory containing matching rknn_api.h")
set(RKNN_LIBRARY "" CACHE FILEPATH "Matching AArch64 librknnrt.so")
if(NOT EXISTS "${RKNN_INCLUDE_DIR}/rknn_api.h" OR NOT EXISTS "${RKNN_LIBRARY}")
  message(FATAL_ERROR "Set RKNN_INCLUDE_DIR and RKNN_LIBRARY explicitly to the installed matching SDK")
endif()
find_path(RGA_INCLUDE_DIR im2d.h PATH_SUFFIXES rga REQUIRED)
find_library(RGA_LIBRARY rga REQUIRED)
add_library(rknnsharp_native SHARED src/rknnsharp.cpp)
# The packaged bridge resolves vendor libraries from the deployment environment,
# never from the private SDK directory used to build it.
set_target_properties(rknnsharp_native PROPERTIES
  SKIP_BUILD_RPATH TRUE
  INSTALL_RPATH ""
  INSTALL_RPATH_USE_LINK_PATH FALSE)
install(TARGETS rknnsharp_native LIBRARY DESTINATION lib)
target_include_directories(rknnsharp_native PUBLIC include PRIVATE "${RKNN_INCLUDE_DIR}" "${RGA_INCLUDE_DIR}")
target_link_libraries(rknnsharp_native PRIVATE "${RKNN_LIBRARY}" "${RGA_LIBRARY}")
target_compile_options(rknnsharp_native PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
add_executable(rknnsharp_hardware_smoke tests/hardware_smoke.cpp)
target_include_directories(rknnsharp_hardware_smoke PRIVATE "${RKNN_INCLUDE_DIR}" "${RGA_INCLUDE_DIR}")
target_link_libraries(rknnsharp_hardware_smoke PRIVATE rknnsharp_native "${RKNN_LIBRARY}" "${RGA_LIBRARY}")
find_package(Threads REQUIRED)
add_executable(rknnsharp_core_benchmark tests/core_benchmark.cpp)
target_include_directories(rknnsharp_core_benchmark PRIVATE "${RKNN_INCLUDE_DIR}" "${RGA_INCLUDE_DIR}")
target_link_libraries(rknnsharp_core_benchmark PRIVATE "${RKNN_LIBRARY}" "${RGA_LIBRARY}" Threads::Threads)
add_executable(rknnsharp_draw_batch_benchmark tests/draw_batch_benchmark.cpp)
target_include_directories(rknnsharp_draw_batch_benchmark PRIVATE include "${RKNN_INCLUDE_DIR}" "${RGA_INCLUDE_DIR}")
target_link_libraries(rknnsharp_draw_batch_benchmark PRIVATE "${RKNN_LIBRARY}" "${RGA_LIBRARY}" Threads::Threads)
add_executable(rknnsharp_run_output_regression tests/run_output_regression.cpp)
target_include_directories(rknnsharp_run_output_regression PRIVATE "${RKNN_INCLUDE_DIR}" "${RGA_INCLUDE_DIR}")
target_link_libraries(rknnsharp_run_output_regression PRIVATE rknnsharp_native "${RGA_LIBRARY}" ${CMAKE_DL_LIBS})
add_executable(rknnsharp_float_output_unit tests/float_output_unit.cpp)
target_include_directories(rknnsharp_float_output_unit PRIVATE src "${RKNN_INCLUDE_DIR}")
target_compile_options(rknnsharp_float_output_unit PRIVATE -Wall -Wextra -Werror)
add_executable(rknnsharp_letterbox_fill_unit tests/letterbox_fill_unit.cpp)
target_include_directories(rknnsharp_letterbox_fill_unit PRIVATE src)
target_compile_options(rknnsharp_letterbox_fill_unit PRIVATE -Wall -Wextra -Werror)
add_executable(rknnsharp_float_mode_regression tests/float_mode_regression.cpp)
target_include_directories(rknnsharp_float_mode_regression PRIVATE src "${RKNN_INCLUDE_DIR}" "${RGA_INCLUDE_DIR}")
target_link_libraries(rknnsharp_float_mode_regression PRIVATE rknnsharp_native "${RGA_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(rknnsharp_float_mode_regression PRIVATE -Wall -Wextra -Werror)
