cmake_minimum_required(VERSION 3.16)
project(rknnsharp_gst C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.20 gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-allocators-1.0)
add_library(rknnsharp_gst SHARED rknnsharp_gst.c)
target_link_libraries(rknnsharp_gst PRIVATE PkgConfig::GST)
target_compile_features(rknnsharp_gst PRIVATE c_std_11)
target_compile_options(rknnsharp_gst PRIVATE -Wall -Wextra -Werror)
add_executable(rknnsharp_gst_probe probe.c)
target_link_libraries(rknnsharp_gst_probe PRIVATE rknnsharp_gst)

include(CTest)
if(BUILD_TESTING)
    add_executable(rknnsharp_gst_colorimetry_regression colorimetry_regression.c)
    target_link_libraries(rknnsharp_gst_colorimetry_regression PRIVATE PkgConfig::GST)
    target_compile_features(rknnsharp_gst_colorimetry_regression PRIVATE c_std_11)
    target_compile_options(rknnsharp_gst_colorimetry_regression PRIVATE -Wall -Wextra -Werror)
    add_test(NAME explicit_colorimetry COMMAND rknnsharp_gst_colorimetry_regression)
endif()
